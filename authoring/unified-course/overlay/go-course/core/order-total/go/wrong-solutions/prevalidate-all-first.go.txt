package lab

import (
 "errors"
 "math"
)

type Line struct {
 UnitCents int64 // 单价：整数“分”，不是“元”的浮点数。
 Quantity int64
}
type Order struct { Lines []Line }
var ErrInvalidLine = errors.New("单价不能为负且数量必须为正")
var ErrTotalOverflow = errors.New("金额超出int64范围")
var _ = math.MaxInt64

func (o Order) TotalCents() (int64, error) {
	// 练习区开始：只替换下方函数体片段 🧪
 for _, line := range o.Lines { if line.UnitCents < 0 || line.Quantity <= 0 {return 0, ErrInvalidLine} }
 var total int64
 for _, line := range o.Lines {
  if line.UnitCents > math.MaxInt64/line.Quantity {return 0, ErrTotalOverflow}
  subtotal:=line.UnitCents*line.Quantity
  if total > math.MaxInt64-subtotal {return 0, ErrTotalOverflow}
  total+=subtotal
 }
 return total,nil
	// 练习区结束
}
