package labs;

/** C11-04：纯策略易于公开测试，实际状态由KubernetesApp采集。 */
public final class ProbePolicy {
    private ProbePolicy() {}
    public static boolean startup(boolean initialized) { return initialized; }
    public static boolean live(boolean localFailure, boolean dependencyHealthy) {
        if (localFailure) return false;
        return true; // 下游故障不是重启本进程的理由。
    }
    public static boolean ready(boolean initialized, boolean dependencyHealthy,
                                boolean seeded, boolean draining, boolean forcedNotReady) {
        if (!initialized || !dependencyHealthy || !seeded) return false;
        if (draining || forcedNotReady) return false;
        return true;
    }
}
